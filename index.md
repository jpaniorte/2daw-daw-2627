<script>
(function () {
  // Resalta la semana (lunes-domingo) en la que cae hoy, no solo los días
  // exactos de sesión: una fila como "Mié 7 oct" debe seguir resaltada
  // el resto de esa semana (jueves, viernes...), no solo el miércoles.
  function startOfWeek(d) {
    var day = d.getDay(); // 0=domingo .. 6=sábado
    var diff = day === 0 ? 6 : day - 1; // días desde el lunes
    var r = new Date(d);
    r.setDate(d.getDate() - diff);
    r.setHours(0, 0, 0, 0);
    return r;
  }
  function endOfWeek(d) {
    var day = d.getDay();
    var diff = day === 0 ? 0 : 7 - day; // días hasta el domingo
    var r = new Date(d);
    r.setDate(d.getDate() + diff);
    r.setHours(23, 59, 59, 999);
    return r;
  }
  var today = new Date();
  today.setHours(0, 0, 0, 0);
  document.querySelectorAll('.cal-table tbody tr[data-start]').forEach(function (tr) {
    var start = startOfWeek(new Date(tr.getAttribute('data-start') + 'T00:00:00'));
    var end = endOfWeek(new Date(tr.getAttribute('data-end') + 'T00:00:00'));
    if (today >= start && today <= end) {
      tr.classList.add('cal-today');
    }
  });

  // Barra de progreso: % de sesiones ya pasadas sobre el total del curso.
  var totalSessions = 0;
  var doneSessions = 0;
  document.querySelectorAll('.cal-table tbody tr[data-start]').forEach(function (tr) {
    var sessions = parseInt(tr.getAttribute('data-sessions'), 10) || 0;
    var end = new Date(tr.getAttribute('data-end') + 'T23:59:59');
    totalSessions += sessions;
    if (today > end) {
      doneSessions += sessions;
    }
  });
  var pct = totalSessions ? Math.round((doneSessions / totalSessions) * 100) : 0;
  var fill = document.getElementById('progress-fill');
  var text = document.getElementById('progress-text');
  if (fill) fill.style.width = pct + '%';
  if (text) {
    text.textContent = 'Progreso del curso: ' + pct + '% (' + doneSessions + ' de ' + totalSessions + ' sesiones)';
  }

  // Fechas de control: tachada si ya pasó; resaltada si es próxima
  // (dentro de los próximos 6 días, sin haber pasado todavía).
  var UPCOMING_DAYS = 6;
  document.querySelectorAll('.ctrl-date[data-control]').forEach(function (el) {
    var controlMidnight = new Date(el.getAttribute('data-control') + 'T00:00:00');
    var controlEnd = new Date(el.getAttribute('data-control') + 'T23:59:59');
    if (today > controlEnd) {
      el.classList.add('passed');
    } else {
      var daysLeft = Math.round((controlMidnight - today) / (1000 * 60 * 60 * 24));
      if (daysLeft <= UPCOMING_DAYS) {
        el.classList.add('upcoming');
      }
    }
  });
})();
</script>
